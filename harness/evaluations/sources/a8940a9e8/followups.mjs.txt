#!/usr/bin/env node
// The follow-up ledger: one file per open item in docs/follow-ups/.
//
//   node scripts/followups.mjs check            validate entries and every reference to them
//   node scripts/followups.mjs new "<title>"    create the next entry and print its path
//   node scripts/followups.mjs index            rewrite the generated list in the README
//
// An entry is deleted in the commit that finishes it, so an id that still appears anywhere in
// the tree after that fails `check`.
import { execFileSync } from "node:child_process";
import { existsSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const ROOT = join(dirname(fileURLToPath(import.meta.url)), "..");
const DIR = join(ROOT, "docs/follow-ups");
const README = join(DIR, "README.md");

const FIELDS = {
  status: ["open", "blocked"],
  priority: ["high", "medium", "low"],
  // What the item is waiting for: a code change, a question answered, a run on a phone,
  // a missing test, another project, or the owner's decision.
  needs: ["fix", "investigation", "device", "test", "upstream", "decision"],
};
const SCANNED = /\.(md|ts|tsx|mjs|cjs|js|sh|py)$/;
const NOT_OURS = [
  ".agents/",
  "app/vendor/",
  "app/modules/bitchat-module/ios/",
  "app/modules/bitchat-module/android/",
];
const ID = /\bF\d{2,4}\b/g;
// The marker has to open the comment, so prose and quoted upstream source that mention one pass.
const BARE_MARKER = /(?:\/\/|\/\*|^\s*\*|^\s*#)\s*(TODO|FIXME|HACK|XXX)\b/;
const LIST_START =
  "<!-- list:start (generated by scripts/followups.mjs index) -->";
const LIST_END = "<!-- list:end -->";
const NEXT_ID = /<!-- next-id: (\d+) -->/;

function readEntries() {
  const entries = [];
  const problems = [];
  for (const name of readdirSync(DIR).sort()) {
    if (!/^F\d+\.md$/.test(name)) continue;
    const text = readFileSync(join(DIR, name), "utf8");
    const where = `docs/follow-ups/${name}`;
    const front = text.match(/^---\n([\s\S]*?)\n---\n([\s\S]*)$/);
    if (!front) {
      problems.push(`${where}: no front matter`);
      continue;
    }
    const meta = {};
    for (const line of front[1].split("\n")) {
      const pair = line.match(/^(\w+): (.*)$/);
      if (pair)
        meta[pair[1]] = pair[2].startsWith('"') ? JSON.parse(pair[2]) : pair[2];
    }
    if (meta.id !== name.replace(".md", ""))
      problems.push(`${where}: id does not match the file name`);
    if (!meta.title || meta.title.length > 120)
      problems.push(`${where}: title is missing or over 120 characters`);
    for (const [field, allowed] of Object.entries(FIELDS)) {
      if (!allowed.includes(meta[field])) {
        problems.push(
          `${where}: ${field} must be one of ${allowed.join(", ")}`,
        );
      }
    }
    const [what, doneWhen] = front[2].split(/^## Done when$/m);
    if (!what.trim()) problems.push(`${where}: says nothing about the work`);
    if (!doneWhen?.trim()) problems.push(`${where}: has no "## Done when"`);
    entries.push({ ...meta, number: Number(name.slice(1, -3)) });
  }
  entries.sort((a, b) => a.number - b.number);
  return { entries, problems };
}

function renderList(entries) {
  const rows = entries.map(
    (e) =>
      `| [${e.id}](${e.id}.md) | ${e.priority} | ${e.needs} | ${e.title.replaceAll("|", "\\|")} |`,
  );
  return [
    "| Id | Priority | Needs | What |",
    "| --- | --- | --- | --- |",
    ...rows,
  ].join("\n");
}

function withList(readme, entries) {
  const start = readme.indexOf(LIST_START);
  const end = readme.indexOf(LIST_END);
  if (start < 0 || end < 0)
    throw new Error("docs/follow-ups/README.md has lost its list markers");
  return `${readme.slice(0, start)}${LIST_START}\n\n${renderList(entries)}\n\n${readme.slice(end)}`;
}

function trackedFiles() {
  const out = execFileSync("git", ["ls-files", "-co", "--exclude-standard"], {
    cwd: ROOT,
    encoding: "utf8",
    maxBuffer: 64 << 20,
  });
  return out
    .split("\n")
    .filter(
      (f) =>
        SCANNED.test(f) &&
        !NOT_OURS.some((p) => f.startsWith(p)) &&
        existsSync(join(ROOT, f)),
    );
}

function check() {
  const { entries, problems } = readEntries();
  const known = new Set(entries.map((e) => e.id));
  const readme = readFileSync(README, "utf8");

  const next = Number(readme.match(NEXT_ID)?.[1]);
  const highest = entries.at(-1)?.number ?? 0;
  if (!(next > highest))
    problems.push(
      `docs/follow-ups/README.md: next-id must be above F${highest}`,
    );
  if (withList(readme, entries) !== readme)
    problems.push(
      "docs/follow-ups/README.md: list is out of date; run `bun run followups index`",
    );

  for (const file of trackedFiles()) {
    const isCode = !file.endsWith(".md");
    readFileSync(join(ROOT, file), "utf8")
      .split("\n")
      .forEach((line, i) => {
        for (const id of line.match(ID) ?? []) {
          if (!known.has(id))
            problems.push(`${file}:${i + 1}: ${id} is not an open follow-up`);
        }
        if (isCode && BARE_MARKER.test(line)) {
          problems.push(
            `${file}:${i + 1}: bare ${line.match(BARE_MARKER)[1]}; write FOLLOWUP(F<n>) and add the entry, or just do it`,
          );
        }
      });
  }

  if (problems.length) {
    console.error(problems.join("\n"));
    console.error(
      `\n${problems.length} follow-up problem(s). See docs/follow-ups/README.md.`,
    );
    process.exit(1);
  }
  console.log(`${entries.length} open follow-ups; every reference resolves.`);
}

function create(title) {
  if (!title)
    throw new Error('usage: followups new "<what is left, in one line>"');
  const readme = readFileSync(README, "utf8");
  const number = Number(readme.match(NEXT_ID)[1]);
  const id = `F${number}`;
  const body = `---\nid: ${id}\ntitle: ${JSON.stringify(title)}\nstatus: open\npriority: medium\nneeds: fix\n---\n\nWhat is wrong or missing, where (name the file and symbol, not a line number), and why it was\nleft.\n\n## Done when\n\nThe observable result that lets someone delete this file.\n`;
  writeFileSync(join(DIR, `${id}.md`), body);
  writeFileSync(
    README,
    readme.replace(NEXT_ID, `<!-- next-id: ${number + 1} -->`),
  );
  index();
  console.log(`docs/follow-ups/${id}.md`);
}

function index() {
  writeFileSync(
    README,
    withList(readFileSync(README, "utf8"), readEntries().entries),
  );
}

const [command, ...rest] = process.argv.slice(2);
({ check, index, new: () => create(rest.join(" ")) })[command ?? "check"]();
